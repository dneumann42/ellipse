#+begin_src nim

  widget CenterBox:
    layout:
      row(stretch = true, align = Center, justify = Center):
        col(stretch = true, align = Center, justiy = Center):
          slot()

  widget Counter:
    state:
      count = 0
    layout:
      button "-":
        discard
      label(count)
      button "+":
        discard  

  widget Application:
    layout:
      CenterBox:
        p "Hello, World!"
